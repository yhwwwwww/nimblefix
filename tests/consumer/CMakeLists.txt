cmake_minimum_required(VERSION 3.23)
project(NimbleFIXConsumer LANGUAGES CXX)
option(CONSUMER_SCHEMA "Test the installed XML generator" ON)
option(CONSUMER_FEATURES "Test optional installed features" OFF)
if(CONSUMER_FEATURES)
    find_package(NimbleFIX 0.4 CONFIG REQUIRED COMPONENTS core tls xml fix44 fix50sp2)
else()
    find_package(NimbleFIX 0.4 CONFIG REQUIRED COMPONENTS core)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE NimbleFIX::nimblefix)
if(CONSUMER_SCHEMA)
    nimblefix_generate_schema(TARGET venue_schema NAMESPACE venue
        HEADER venue.hpp XML_FILES "${CMAKE_CURRENT_SOURCE_DIR}/venue.xml")
    add_executable(schema-consumer typed.cpp)
    target_link_libraries(schema-consumer PRIVATE venue_schema)
endif()
if(CONSUMER_FEATURES)
    add_executable(features fix44_features.cpp)
    target_link_libraries(features PRIVATE NimbleFIX::nimblefix NimbleFIX::fix44)
    add_executable(xml-runtime xml_runtime.cpp)
    target_link_libraries(xml-runtime PRIVATE NimbleFIX::nimblefix_xml)
endif()
